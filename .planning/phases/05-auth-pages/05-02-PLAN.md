Create RegisterPage.vue in src/frontend/src/pages/auth/ directory.

Component structure:
1. Template:
   - Centered card layout using q-page with flexbox centering
   - Mobile-responsive: col-12 col-sm-10 col-md-8 col-lg-6 (wider than login due to more fields)
   - q-card with q-card-section for header (t('auth.register')) and form
   - q-form with @submit.prevent calling handleRegister

   Required fields (each with lazy-rules, :error, :error-message bindings):
   - q-input login: v-model="form.login", :label="t('auth.login') or 'Username'", :rules="[required, maxLen50]"
   - q-input email: v-model="form.email", type="email", :label="t('auth.email')", :rules="[required, validEmail, minLen5, maxLen100]"
   - q-input firstName: v-model="form.firstName", :label="t('auth.firstName')", :rules="[required, maxLen50]"
   - q-input lastName: v-model="form.lastName", :label="t('auth.lastName')", :rules="[required, maxLen50]"
   - q-input password: v-model="form.password", :type="isPwd ? 'password' : 'text'", :label="t('auth.password')", :rules="[required, minLen5, maxLen100]", with visibility toggle icon
   - q-input confirmPassword: v-model="form.confirmPassword", :type="isPwd2 ? 'password' : 'text'", :label="t('auth.confirmPassword')", :rules="[required, passwordMatch]", with visibility toggle icon

   Optional fields:
   - q-input phone: v-model="form.phone", type="tel", :label="t('auth.phone')"
   - q-input dob: v-model="form.dob", type="date", :label="t('auth.dateOfBirth')", :rules="[pastDate]" (only validate if filled)
   - q-select gender: v-model="form.gender", :options="genderOptions", :label="t('auth.gender')", emit-value, map-options
   - q-checkbox otpEnabled: v-model="form.otpEnabled", label="Enable Two-Factor Authentication"

   Layout: Use q-row and q-col for responsive grid:
   - First row: login (col-12 col-md-6), email (col-12 col-md-6)
   - Second row: firstName (col-12 col-md-6), lastName (col-12 col-md-6)
   - Third row: password (col-12 col-md-6), confirmPassword (col-12 col-md-6)
   - Fourth row: phone (col-12 col-md-6), dob (col-12 col-md-6)
   - Fifth row: gender (col-12 col-md-6), otpEnabled checkbox (col-12 col-md-6)

   Error display: q-banner type="negative" if hasError and not isValidationError
   Submit button: q-btn type="submit", :loading="isSubmitting", :disable="isSubmitting", :label="t('auth.createAccount')", full width
   Link to login: t('auth.haveAccount') + router-link to /login

2. Script setup:
   - Imports: ref, computed, useRouter, useQuasar, useI18n, accountApi, useErrorHandler
   - const router = useRouter(); const $q = useQuasar(); const { t } = useI18n();
   - const { setError, clearError, hasError, errorMessage, isValidationError, hasFieldError, getFieldError } = useErrorHandler();

   Form state:
   const form = ref({
     login: '',
     email: '',
     password: '',
     confirmPassword: '',
     firstName: '',
     lastName: '',
     phone: '',
     dob: '',
     gender: null,
     otpEnabled: false
   });
   const isPwd = ref(true);
   const isPwd2 = ref(true);
   const isSubmitting = ref(false);

   Gender options:
   const genderOptions = [
     { label: t('auth.male'), value: 'MALE' },
     { label: t('auth.female'), value: 'FEMALE' },
     { label: t('auth.other'), value: 'OTHER' }
   ];

   Validation rules:
   - required = val => !!val || t('validation.required')
   - validEmail = val => /.+@.+\..+/.test(val) || t('validation.email')
   - minLen5 = val => val.length >= 5 || t('validation.minLength', { min: 5 })
   - maxLen50 = val => val.length <= 50 || t('validation.maxLength', { max: 50 })
   - maxLen100 = val => val.length <= 100 || t('validation.maxLength', { max: 100 })
   - passwordMatch = val => val === form.value.password || t('validation.passwordMatch')
   - pastDate = val => !val || new Date(val) < new Date() || t('validation.pastDate')

   handleRegister async function:
   a. clearError()
   b. isSubmitting.value = true
   c. Build userData object from form, only including non-empty optional fields
   d. try { await accountApi.register(userData); }
   e. Show success notification: $q.notify({ type: 'positive', message: t('success.registered') })
   f. Redirect: router.push('/login')
   g. catch (err) { setError(err); }
   h. finally { isSubmitting.value = false; }

3. Styling:
   - Use Quasar utility classes
   - Card should have max-width: 600px
   - Use q-gutter-md on rows for spacing
